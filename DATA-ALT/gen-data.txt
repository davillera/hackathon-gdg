def generate_time_series(catalog, days=730):
    np.random.seed(101)
    end_date = datetime.date.today() - datetime.timedelta(days=1)
    start_date = end_date - datetime.timedelta(days=days - 1)
    date_range = pd.date_range(start=start_date, end=end_date, freq="D")
    
    historical_records = []

    for item in catalog:
        base = item["base_consumo"]
        t_slope = item["tendencia_anual"] / 365.0

        for idx, current_dt in enumerate(date_range):
            weekday = current_dt.weekday() # 0 = Lunes, 6 = Domingo
            
            # 1. Multiplicadores por día de la semana (Producción industrial)
            if weekday in [0, 1, 2, 3]:   # Lun - Jue: operación full
                day_factor = 1.05 + np.random.uniform(-0.05, 0.05)
            elif weekday == 4:            # Viernes: cierre de turno temprano
                day_factor = 0.90 + np.random.uniform(-0.04, 0.04)
            elif weekday == 5:            # Sábado: turno parcial de mantenimiento
                day_factor = 0.35 + np.random.uniform(-0.03, 0.03)
            else:                         # Domingo: planta cerrada
                day_factor = 0.03 + np.random.uniform(0.00, 0.02)
                
            # 2. Multiplicadores mensuales (ej. picos de producción en Q3/Q4)
            month = current_dt.month
            month_factor = 1.0 + (0.12 * np.sin(2 * np.pi * (month - 3) / 12))
            
            # 3. Tendencia lineal acumulada
            trend_factor = 1.0 + (t_slope * idx)
            
            # 4. Ruido gaussiano (~10% de dispersión)
            noise = np.random.normal(0, base * 0.08)
            
            # 5. Choque de consumo puntual ocasional (evento de mantenimiento mayor cada ~60 días)
            shock = (base * 0.8) if (np.random.rand() < 0.015 and weekday < 5) else 0.0

            consumo = (base * day_factor * month_factor * trend_factor) + noise + shock
            consumo_final = round(max(0.0, float(consumo)), 2)

            historical_records.append({
                "fecha": current_dt.date(),
                "id_insumo": item["id_insumo"],
                "cantidad_consumida": consumo_final,
                "tipo_turno": "TOTAL",
                "es_feriado": False
            })

    return pd.DataFrame(historical_records)

def main():
    print(f"Generando histórico sintético para {len(CATALOG)} insumos...")
    df_hist = generate_time_series(CATALOG, days=730)
    
    df_curr = pd.DataFrame([
        {
            "id_insumo": it["id_insumo"],
            "nombre": it["nombre"],
            "categoria": it["categoria"],
            "unidad": it["unidad"],
            "stock_actual": float(it["stock_actual"]),
            "stock_minimo": float(it["stock_minimo"]),
            "costo_unitario_usd": float(it["costo_unitario_usd"]),
            "lead_time_dias": int(it["lead_time_dias"]),
            "ultima_actualizacion": datetime.datetime.now(datetime.timezone.utc)
        }
        for it in CATALOG
    ])

    # Carga a BigQuery
    hist_table_id = f"{PROJECT_ID}.{DATASET_ID}.historical_inventory"
    curr_table_id = f"{PROJECT_ID}.{DATASET_ID}.current_inventory"

    job_cfg = bigquery.LoadJobConfig(write_disposition=bigquery.WriteDisposition.WRITE_TRUNCATE)

    print(f"Cargando {len(df_hist)} registros en `{hist_table_id}`...")
    client.load_table_from_dataframe(df_hist, hist_table_id, job_config=job_cfg).result()

    print(f"Cargando {len(df_curr)} registros en `{curr_table_id}`...")
    client.load_table_from_dataframe(df_curr, curr_table_id, job_config=job_cfg).result()

    print("Carga completa y verificada.")

if __name__ == "__main__":
    main()
